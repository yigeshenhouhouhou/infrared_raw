cmake_minimum_required(VERSION 3.15)
project(InfraredRaw LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
file(GLOB_RECURSE RAW_SOURCES CONFIGURE_DEPENDS vendor/LibRaw-0.22.3/src/*.cpp)
list(APPEND RAW_SOURCES vendor/LibRaw-0.22.3/internal/dcraw_common.cpp)
list(FILTER RAW_SOURCES EXCLUDE REGEX "dcraw_common.cpp$")
add_library(raw STATIC ${RAW_SOURCES})
target_include_directories(raw PUBLIC vendor/LibRaw-0.22.3)
target_include_directories(raw PRIVATE vendor/libjpeg-turbo-3.2.0/src ${JPEG_BINARY_DIR})
if(ZLIB_INCLUDE_DIR)
  target_include_directories(raw PRIVATE ${ZLIB_INCLUDE_DIR})
endif()
target_compile_definitions(raw PRIVATE LIBRAW_NODLL USE_ZLIB USE_JPEG)
if(ANDROID)
  target_link_libraries(raw PRIVATE ${JPEG_LIBRARY})
else()
  # MinGW's linker cannot open Unicode absolute paths. Use a relative input.
  file(RELATIVE_PATH JPEG_RELATIVE "${CMAKE_BINARY_DIR}" "${JPEG_LIBRARY}")
  target_link_libraries(raw PRIVATE "-Wl,${JPEG_RELATIVE}")
endif()
target_compile_options(raw PRIVATE -O2 -w)
if(ANDROID)
  add_library(infrared SHARED native/core.cpp native/color.cpp native/bridge.cpp)
  target_link_libraries(infrared PRIVATE raw z jnigraphics log)
  target_link_options(infrared PRIVATE "-Wl,-z,max-page-size=16384")
else()
  add_executable(native_check native/core.cpp native/color.cpp native/check.cpp)
  target_link_libraries(native_check PRIVATE raw ${ZLIB_LIBRARY} ws2_32)
endif()
